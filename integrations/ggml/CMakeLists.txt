cmake_minimum_required(VERSION 3.20)
project(ogpu_ggml_consumer LANGUAGES C CXX)
if(APPLE)
    set(FOUNDATION_DEFAULT OFF)
else()
    set(FOUNDATION_DEFAULT ON)
endif()
option(OGPU_GGML_FOUNDATION "Use the explicit foundation contract (Linux Vulkan only)" ${FOUNDATION_DEFAULT})
if(OGPU_GGML_FOUNDATION AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Foundation consumer currently supports Linux Vulkan only")
endif()
set(GGML_SOURCE "" CACHE PATH "Clean GGML checkout at the pinned revision")
set(OGPU_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../target/release/${CMAKE_SHARED_LIBRARY_PREFIX}ogpu${CMAKE_SHARED_LIBRARY_SUFFIX}" CACHE FILEPATH "Matching OGPU shared library")
set(OGPU_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../include" CACHE PATH "Matching public OGPU headers")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GGML_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE GGML_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT GGML_REV STREQUAL "7840aaba1989c6deeefede1d77d5aaf8f52b947e")
    message(FATAL_ERROR "Wrong GGML revision: ${GGML_REV}")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${GGML_SOURCE}" diff --exit-code HEAD --
    COMMAND_ERROR_IS_FATAL ANY)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" ggml)
add_library(ogpu SHARED IMPORTED)
set_target_properties(ogpu PROPERTIES IMPORTED_LOCATION "${OGPU_LIBRARY}")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # Rust's local cdylib has no SONAME. Use -L/-l plus CMake's build RPATH,
    # rather than embedding the supplied absolute library path in DT_NEEDED.
    set_target_properties(ogpu PROPERTIES IMPORTED_NO_SONAME TRUE)
endif()
add_executable(ogpu-mnist mnist.cpp backend.cpp "${GGML_SOURCE}/examples/mnist/mnist-common.cpp")
target_compile_features(ogpu-mnist PRIVATE cxx_std_17)
target_compile_options(ogpu-mnist PRIVATE -Wall -Wextra -Werror)
target_include_directories(ogpu-mnist PRIVATE "${OGPU_INCLUDE_DIR}" "${GGML_SOURCE}/src" "${GGML_SOURCE}/examples/mnist")
target_link_libraries(ogpu-mnist PRIVATE ogpu ggml ggml-cpu)
add_executable(ogpu-ggml-lifecycle lifecycle.cpp backend.cpp)
target_compile_features(ogpu-ggml-lifecycle PRIVATE cxx_std_17)
target_compile_options(ogpu-ggml-lifecycle PRIVATE -Wall -Wextra -Werror)
target_include_directories(ogpu-ggml-lifecycle PRIVATE "${OGPU_INCLUDE_DIR}" "${GGML_SOURCE}/src")
target_link_libraries(ogpu-ggml-lifecycle PRIVATE ogpu ggml ggml-cpu)
add_executable(ogpu-ggml-convert convert-weights.cpp)
target_compile_features(ogpu-ggml-convert PRIVATE cxx_std_17)
target_compile_options(ogpu-ggml-convert PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ogpu-ggml-convert PRIVATE ggml)
add_executable(ogpu-ggml-matrix-check matrix-check.cpp backend.cpp)
target_compile_features(ogpu-ggml-matrix-check PRIVATE cxx_std_17)
target_compile_options(ogpu-ggml-matrix-check PRIVATE -Wall -Wextra -Werror)
target_include_directories(ogpu-ggml-matrix-check PRIVATE "${OGPU_INCLUDE_DIR}" "${GGML_SOURCE}/src")
target_link_libraries(ogpu-ggml-matrix-check PRIVATE ogpu ggml ggml-cpu)
foreach(consumer ogpu-mnist ogpu-ggml-lifecycle ogpu-ggml-matrix-check)
    target_compile_definitions(${consumer} PRIVATE OGPU_GGML_FOUNDATION=$<BOOL:${OGPU_GGML_FOUNDATION}>)
endforeach()
if(OGPU_GGML_FOUNDATION)
    add_executable(ogpu-ggml-foundation-check foundation-check.cpp)
    target_compile_features(ogpu-ggml-foundation-check PRIVATE cxx_std_17)
    target_compile_options(ogpu-ggml-foundation-check PRIVATE -Wall -Wextra -Werror)
    target_include_directories(ogpu-ggml-foundation-check PRIVATE "${OGPU_INCLUDE_DIR}")
    target_link_libraries(ogpu-ggml-foundation-check PRIVATE ogpu)
endif()
